\begin{algorithm}[t]
\caption{Rapidly-exploring Random Tree (RRT), user-chosen samples}
\begin{algorithmic}[1]
\Function{RRT}{$q_{\text{start}}, q_{\text{goal}}, \Delta$}
  \State $T \gets \{q_{\text{start}}\}$
  \Repeat
    \State $q_{\text{rand}} \gets \textsc{PointClickedByUser}$
    \State $q_{\text{near}} \gets \Call{Nearest}{T, q_{\text{rand}}}$
    \State $q_{\text{new}} \gets \Call{Steer}{q_{\text{near}}, q_{\text{rand}}, \Delta}$
    \If{$\neg$\Call{CollisionFree}{$q_{\text{near}}, q_{\text{new}}$}}
      \State \textbf{continue}
    \EndIf
    \State $T \gets T \cup \{q_{\text{new}}\}$
    \If{$\lVert q_{\text{new}} - q_{\text{goal}} \rVert \le \Delta$ and \Call{CollisionFree}{$q_{\text{new}}, q_{\text{goal}}$}}
      \State $T \gets T \cup \{q_{\text{goal}}\}$
      \State \Return \Call{ReconstructPath}{$T, \text{parent}, q_{\text{goal}}$}
    \EndIf
  \Until{false}
\EndFunction
\end{algorithmic}
\end{algorithm}
