\begin{algorithm}[t]
\caption{Random walk}
\begin{algorithmic}[1]
\Function{RandomWalk}{$q_{\text{start}}, q_{\text{goal}}, \Delta$}
  \State $q \gets q_{\text{start}}$;\ \ $P \gets (q_{\text{start}})$
  \Repeat
    \State $\theta \gets \textsc{RandomAngle}$
    \State $q_{\text{new}} \gets q + \Delta\,(\cos\theta, \sin\theta)$
    \If{$\neg$\Call{CollisionFree}{$q, q_{\text{new}}$}}
      \State \textbf{continue}
    \EndIf
    \State $q \gets q_{\text{new}}$;\ \ \Call{Append}{$P, q$}
    \If{$\lVert q - q_{\text{goal}} \rVert \le \Delta$ and \Call{CollisionFree}{$q, q_{\text{goal}}$}}
      \State \Return \Call{Append}{$P, q_{\text{goal}}$}
    \EndIf
  \Until{false}
\EndFunction
\end{algorithmic}
\end{algorithm}
